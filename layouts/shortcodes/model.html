{{- /* a 3D model (.stl, .3mf or .obj) from the page bundle, shown as a figure that can be rotated (assets/js/model.js, with partials/model-head.html); without JS, or if it can't load, the stage just offers the file. color: any CSS colour, or a CSS variable (--name or var(--name), which follows the theme), default --model-color (assets/css/model.css, as are the other variables a theme can set). ratio: the stage's aspect-ratio, default 4/3. rotation: where the view starts, "spin,tilt" in degrees (spin round the model from its front, tilt above it), default 35,28. modes: comma-separated from solid, flat, wireframe, xray and normals, first one showing, default solid; two or more get buttons to switch. */ -}} {{- $src := .Get "src" -}} {{- $res := .Page.Resources.Get $src -}} {{- if not $res }}{{ errorf "%s: model %q not found in the page bundle" .Position $src }}{{ end -}} {{- $format := strings.TrimPrefix "." (path.Ext $src) | lower -}} {{- if not (in (slice "stl" "3mf" "obj") $format) }}{{ errorf "%s: model %q is not an .stl, .3mf or .obj" .Position $src }}{{ end -}} {{- $style := slice (printf "aspect-ratio:%s" (.Get "ratio" | default "4/3")) -}} {{- with .Get "color" }}{{ $style = $style | append (printf "--model-color:%s" (cond (hasPrefix . "--") (printf "var(%s)" .) .)) }}{{ end -}} {{- $rotation := .Get "rotation" | default "35,28" | replaceRE `\s` "" -}} {{- if not (findRE `^-?[0-9.]+,-?[0-9.]+$` $rotation) }}{{ errorf "%s: model rotation %q is not \"spin,tilt\" in degrees" .Position $rotation }}{{ end -}} {{- $modes := slice -}} {{- range split (.Get "modes" | default "solid") "," -}} {{- $mode := trim . " " -}} {{- if not (in (slice "solid" "flat" "wireframe" "xray" "normals") $mode) }}{{ errorf "%s: unknown model mode %q" $.Position $mode }}{{ end -}} {{- $modes = $modes | append $mode -}} {{- end -}}
